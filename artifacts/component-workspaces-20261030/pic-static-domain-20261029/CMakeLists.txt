cmake_minimum_required(VERSION 3.24)
project(StaticDomainSensitivity LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(MFEM 4.10 CONFIG REQUIRED)
set(capture /home/treefor/projects/runtimes/pic-cuda-gather-investigation-20261029/source)
set(sealed /home/treefor/projects/runtimes/mfem-gpu-pic-20261029/source)
set(gpu /home/treefor/projects/runtimes/mfem-cuda-benchmark-20261029/cuda-libs/nvidia)
foreach(target static_domain_cpu static_domain_gpu)
 add_executable(${target} source/static_domain.cpp ${capture}/batch_snapshot.cpp)
 target_include_directories(${target} PRIVATE source ${capture} ${sealed}/include)
 target_link_libraries(${target} PRIVATE mfem)
 target_compile_options(${target} PRIVATE -ffp-contract=off)
endforeach()
target_compile_definitions(static_domain_gpu PRIVATE STATIC_DOMAIN_WITH_CUDSS=1)
target_include_directories(static_domain_gpu PRIVATE ${gpu}/cuda_runtime/include ${gpu}/cuda_nvcc/include ${gpu}/cu12/include)
target_link_libraries(static_domain_gpu PRIVATE ${gpu}/cu12/lib/libcudss.so.0 ${gpu}/cuda_runtime/lib/libcudart.so.12)
set_target_properties(static_domain_gpu PROPERTIES BUILD_RPATH "${gpu}/cu12/lib;${gpu}/cublas/lib;${gpu}/cuda_runtime/lib")
