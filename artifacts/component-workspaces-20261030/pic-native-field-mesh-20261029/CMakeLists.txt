cmake_minimum_required(VERSION 3.24)
project(NativePicFieldMesh LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(MFEM 4.10 CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(native_field_core STATIC source/src/mfem_gap_pic.cpp)
target_include_directories(native_field_core PUBLIC source/include PRIVATE source/src)
target_link_libraries(native_field_core PUBLIC mfem OpenSSL::Crypto)
target_compile_options(native_field_core PRIVATE -ffp-contract=off)
add_executable(heating-history-field-mesh source/examples/mfem_heating_history_pic.cpp)
target_include_directories(heating-history-field-mesh PRIVATE source/src)
target_link_libraries(heating-history-field-mesh PRIVATE native_field_core)
target_compile_options(heating-history-field-mesh PRIVATE -ffp-contract=off)
add_executable(test_native_foundation source/tests/test_mfem_gap_pic.cpp)
target_link_libraries(test_native_foundation PRIVATE native_field_core)
target_compile_options(test_native_foundation PRIVATE -ffp-contract=off)
add_executable(test_q1_scalar_gather source/tests/test_mfem_q1_scalar_gather.cpp)
target_link_libraries(test_q1_scalar_gather PRIVATE native_field_core)
target_compile_options(test_q1_scalar_gather PRIVATE -ffp-contract=off)
add_library(reaction_unchanged STATIC reaction-core/src/reaction_transactions.cpp source/src/mfem_pic_reaction_batch.cpp)
target_include_directories(reaction_unchanged PUBLIC reaction-core/include source/include vendor)
target_compile_definitions(reaction_unchanged PRIVATE PIC_REACTION_NATIVE_SOURCE_SHA256="${PIC_REACTION_NATIVE_SOURCE_SHA256}")
target_link_libraries(reaction_unchanged PUBLIC native_field_core)
target_compile_options(reaction_unchanged PRIVATE -std=c++17 -O2 -ffp-contract=off)
add_executable(test_native_reactions tests/test_native_reactions.cpp)
target_include_directories(test_native_reactions PRIVATE tests)
target_link_libraries(test_native_reactions PRIVATE reaction_unchanged)
target_compile_options(test_native_reactions PRIVATE -std=c++17 -O2 -ffp-contract=off)
add_executable(test_native_field_mesh tests/test_native_field_mesh.cpp)
target_include_directories(test_native_field_mesh PRIVATE source/include source/src vendor)
target_link_libraries(test_native_field_mesh PRIVATE mfem OpenSSL::Crypto)
target_compile_options(test_native_field_mesh PRIVATE -ffp-contract=off)
