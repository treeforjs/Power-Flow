# Eigen's SparseLU backs the optional direct Newton solve (solver.linear_solver "direct"); without it that choice rejects.
find_package(Eigen3 3.3 CONFIG QUIET)
if(Eigen3_FOUND)
  # header only: take its include path, not the target, so the static library's export does not require Eigen downstream
  get_target_property(_raikou_eigen_include Eigen3::Eigen INTERFACE_INCLUDE_DIRECTORIES)
  target_include_directories(raikou SYSTEM PRIVATE $<BUILD_INTERFACE:${_raikou_eigen_include}>)
  target_compile_definitions(raikou PRIVATE RAIKOU_HAS_EIGEN)
  message(STATUS "Raikou: sparse direct solver from Eigen ${Eigen3_VERSION}")
endif()

if(RAIKOU_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(_raikou_cuda_runtime_license "${RAIKOU_CUDA_RUNTIME_LICENSE}")
  if(NOT _raikou_cuda_runtime_license)
    get_filename_component(_raikou_cuda_root "${CUDAToolkit_BIN_DIR}" DIRECTORY)
    string(REGEX MATCH "^[0-9]+\\.[0-9]+" _raikou_cuda_release "${CUDAToolkit_VERSION}")
    string(REPLACE "." "-" _raikou_cuda_package "${_raikou_cuda_release}")
    foreach(_candidate IN ITEMS
        "${_raikou_cuda_root}/licenses/cuda_cudart/LICENSE"
        "${CUDAToolkit_LIBRARY_ROOT}/licenses/cuda_cudart/LICENSE"
        "/usr/share/doc/cuda-cudart-${_raikou_cuda_package}/copyright"
        "${_raikou_cuda_root}/EULA.txt")
      if(EXISTS "${_candidate}")
        set(_raikou_cuda_runtime_license "${_candidate}")
        break()
      endif()
    endforeach()
  endif()
  if(NOT EXISTS "${_raikou_cuda_runtime_license}")
    message(FATAL_ERROR "CUDA builds must package the matching runtime license. Set RAIKOU_CUDA_RUNTIME_LICENSE to the full license supplied with this toolkit.")
  endif()
  target_sources(raikou PRIVATE src/backend_cuda.cu src/device_cuda.cu)
  if(RAIKOU_BUILD_RUNTIME)
    target_sources(raikou PRIVATE src/chemistry_device_cuda.cu)
  endif()
  target_compile_definitions(raikou PRIVATE RAIKOU_HAS_CUDA)
  target_link_libraries(raikou PUBLIC CUDA::cudart_static)
  set_target_properties(raikou PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
endif()
