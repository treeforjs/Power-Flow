cmake_minimum_required(VERSION 3.24)
project(EmissionIntegrationQualification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT DEFINED snapshot)
  set(snapshot "${CMAKE_CURRENT_SOURCE_DIR}/../snapshots/usable-api-v1")
endif()
if(EXISTS "${snapshot}/lib/libsurface_emission.a")
  set(emission_library "${snapshot}/lib/libsurface_emission.a")
else()
  set(emission_library "${snapshot}/build/libsurface_emission.a")
endif()
add_executable(integration_qualification integration_qualification.cpp)
target_include_directories(integration_qualification PRIVATE "${snapshot}/include" "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/ss304")
target_link_libraries(integration_qualification PRIVATE "${emission_library}")
target_compile_options(integration_qualification PRIVATE -O3 -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
add_executable(canonical_barrier_diagnostic canonical_barrier_diagnostic.cpp)
target_include_directories(canonical_barrier_diagnostic PRIVATE "${snapshot}/include")
target_link_libraries(canonical_barrier_diagnostic PRIVATE "${emission_library}")
target_compile_options(canonical_barrier_diagnostic PRIVATE -O3 -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
