cmake_minimum_required(VERSION 3.24)
project(SurfaceEmission LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(surface_emission STATIC src/surface_emission.cpp src/micro_layer.cpp)
target_include_directories(surface_emission PUBLIC include)
target_compile_options(surface_emission PRIVATE -O3 -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
add_executable(emission_qualification tests/qualification.cpp)
target_link_libraries(emission_qualification PRIVATE surface_emission)
target_compile_options(emission_qualification PRIVATE -O3 -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
add_executable(emission_micro_qualification tests/micro_qualification.cpp)
target_link_libraries(emission_micro_qualification PRIVATE surface_emission)
target_compile_options(emission_micro_qualification PRIVATE -O3 -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
