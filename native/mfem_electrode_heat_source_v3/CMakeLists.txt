cmake_minimum_required(VERSION 3.24)
project(electrode_heat_source_v3 LANGUAGES CXX)
set(ELECTRODE_HEAT_SOURCE_V2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../mfem_electrode_heat_feedback_v2" CACHE PATH "Exact sealed heat/source-v2 prerequisite")
if(NOT TARGET electrode-heat-feedback-v2)
  add_subdirectory(${ELECTRODE_HEAT_SOURCE_V2_DIR} ${CMAKE_CURRENT_BINARY_DIR}/v2-owner)

option(ELECTRODE_HEAT_SOURCE_V3_BUILD_TESTS "Build sealed native component qualification" ${PROJECT_IS_TOP_LEVEL})
if(ELECTRODE_HEAT_SOURCE_V3_BUILD_TESTS)

target_link_libraries(electrode-heat-source-v3-tests PRIVATE electrode-heat-source-v3 source-v3-native-support)
target_compile_options(electrode-heat-source-v3-tests PRIVATE -O2 -ffp-contract=off -fno-fast-math)
endif()
