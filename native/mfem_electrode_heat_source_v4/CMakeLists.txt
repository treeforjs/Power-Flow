cmake_minimum_required(VERSION 3.24)
project(electrode_heat_source_v4 LANGUAGES CXX)
set(ELECTRODE_HEAT_SOURCE_V3_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../mfem_electrode_heat_source_v3" CACHE PATH "Immutable v3 prerequisite")
if(NOT DEFINED ELECTRODE_HEAT_SOURCE_V3_BUILD_TESTS)
 set(ELECTRODE_HEAT_SOURCE_V3_BUILD_TESTS OFF CACHE BOOL "")
endif()
if(NOT TARGET electrode-heat-source-v3)
 add_subdirectory(${ELECTRODE_HEAT_SOURCE_V3_DIR} ${CMAKE_CURRENT_BINARY_DIR}/v3-owner)
endif()
set(EXACT_PARTICLE_COUNT_HEADER "" CACHE FILEPATH "Existing sealed ExactParticleCount header")
if(NOT EXISTS "${EXACT_PARTICLE_COUNT_HEADER}")
 message(FATAL_ERROR "Provide the existing sealed ExactParticleCount header")
endif()
file(SHA256 "${EXACT_PARTICLE_COUNT_HEADER}" COUNT_SHA)
if(NOT COUNT_SHA STREQUAL "46e01fe22e67cc9deb0da9320edef4041f4b3ef433de8ce1eca484463b405501")
 message(FATAL_ERROR "ExactParticleCount prerequisite differs from source identity b8a9a27")
endif()

# Verify the actual included targets, including parent-supplied prerequisite targets.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/prerequisites.json" PREREQUISITES)
foreach(VERSION IN ITEMS v2 v3)
 if(VERSION STREQUAL "v2")
  get_target_property(PREREQUISITE_ROOT electrode-heat-feedback-v2 SOURCE_DIR)
 else()
  get_target_property(PREREQUISITE_ROOT electrode-heat-source-v3 SOURCE_DIR)
 endif()
 string(JSON ENTRY_COUNT LENGTH "${PREREQUISITES}" "${VERSION}")
 math(EXPR LAST_ENTRY "${ENTRY_COUNT}-1")
 foreach(INDEX RANGE 0 ${LAST_ENTRY})
  string(JSON RELATIVE_PATH GET "${PREREQUISITES}" "${VERSION}" ${INDEX} path)
  string(JSON EXPECTED_SHA GET "${PREREQUISITES}" "${VERSION}" ${INDEX} sha256)
  file(SHA256 "${PREREQUISITE_ROOT}/${RELATIVE_PATH}" ACTUAL_SHA)
  if(NOT ACTUAL_SHA STREQUAL EXPECTED_SHA)
   message(FATAL_ERROR "Sealed ${VERSION} prerequisite changed: ${RELATIVE_PATH}")
  endif()
 endforeach()
endforeach()

find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(electrode-heat-source-v4 INTERFACE)
target_compile_features(electrode-heat-source-v4 INTERFACE cxx_std_20)
target_compile_options(electrode-heat-source-v4 INTERFACE -ffp-contract=off -fno-fast-math)
target_include_directories(electrode-heat-source-v4 INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(electrode-heat-source-v4 INTERFACE HEAT_SOURCE_EXACT_COUNT_HEADER="${EXACT_PARTICLE_COUNT_HEADER}")
target_link_libraries(electrode-heat-source-v4 INTERFACE electrode-heat-source-v3 OpenSSL::Crypto)
option(ELECTRODE_HEAT_SOURCE_V4_BUILD_TESTS "Build focused corrective component tests" ${PROJECT_IS_TOP_LEVEL})
if(ELECTRODE_HEAT_SOURCE_V4_BUILD_TESTS)
 add_library(source-v4-native-support STATIC ${ELECTRODE_HEAT_SOURCE_V3_DIR}/hall-owner/src/mfem_gap_pic.cpp)
 target_include_directories(source-v4-native-support PRIVATE ${ELECTRODE_HEAT_SOURCE_V3_DIR}/hall-owner/include ${ELECTRODE_HEAT_SOURCE_V3_DIR}/hall-owner/src)
 target_link_libraries(source-v4-native-support PRIVATE electrode-heat-source-v3)
 target_compile_options(source-v4-native-support PRIVATE -O2 -ffp-contract=off -fno-fast-math)
 add_executable(electrode-heat-source-v4-tests test_source_owner.cpp)
 target_include_directories(electrode-heat-source-v4-tests PRIVATE ${ELECTRODE_HEAT_SOURCE_V3_DIR}/hall-owner/include ${ELECTRODE_HEAT_SOURCE_V3_DIR}/hall-owner/src)
 target_link_libraries(electrode-heat-source-v4-tests PRIVATE electrode-heat-source-v4 source-v4-native-support)
 target_compile_options(electrode-heat-source-v4-tests PRIVATE -O2 -ffp-contract=off -fno-fast-math)
endif()
