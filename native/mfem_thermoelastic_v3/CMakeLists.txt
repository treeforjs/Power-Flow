cmake_minimum_required(VERSION 3.20)
project(ss304_thermoelastic_v3 LANGUAGES CXX)
find_package(MFEM 4.10 CONFIG REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
add_library(ss304-thermoelastic-v3 STATIC thermoelastic_v3.cpp)
target_include_directories(ss304-thermoelastic-v3 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ss304-thermoelastic-v3 PRIVATE mfem MPI::MPI_CXX)
target_compile_features(ss304-thermoelastic-v3 PUBLIC cxx_std_17)
add_executable(ss304-thermoelastic-v3-server thermoelastic_v3_server.cpp)
target_link_libraries(ss304-thermoelastic-v3-server PRIVATE ss304-thermoelastic-v3 mfem MPI::MPI_CXX)
target_compile_features(ss304-thermoelastic-v3-server PRIVATE cxx_std_17)
add_executable(ss304-thermoelastic-v3-tests test_thermoelastic_v3.cpp)
target_link_libraries(ss304-thermoelastic-v3-tests PRIVATE ss304-thermoelastic-v3 mfem MPI::MPI_CXX)
target_compile_features(ss304-thermoelastic-v3-tests PRIVATE cxx_std_17)
enable_testing()
add_test(NAME ss304-thermoelastic-v3-tests COMMAND ss304-thermoelastic-v3-tests)
