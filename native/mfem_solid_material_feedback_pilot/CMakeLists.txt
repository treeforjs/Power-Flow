cmake_minimum_required(VERSION 3.24)
project(solid_material_feedback_pilot LANGUAGES CXX)
find_package(MFEM 4.10 CONFIG REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
add_executable(solid-material-feedback-pilot pilot.cpp)
target_link_libraries(solid-material-feedback-pilot PRIVATE mfem MPI::MPI_CXX)
target_compile_features(solid-material-feedback-pilot PRIVATE cxx_std_20)
target_compile_definitions(solid-material-feedback-pilot PRIVATE PILOT_SCHEDULE="${CMAKE_CURRENT_SOURCE_DIR}/inputs/schedule.csv")
target_compile_options(solid-material-feedback-pilot PRIVATE -O3 -ffp-contract=off -fno-fast-math)
