cmake_minimum_required(VERSION 3.24)
project(electrode_heat_feedback_v2 LANGUAGES CXX)
find_package(MFEM 4.10 CONFIG REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
set(ELECTRODE_HEAT_SOURCE_TARGET "" CACHE STRING "Existing source-owner target, only when its sealed kernel matches binding_seals.hpp")
if(ELECTRODE_HEAT_SOURCE_TARGET)
  if(NOT TARGET ${ELECTRODE_HEAT_SOURCE_TARGET})
    message(FATAL_ERROR "Selected source-owner target does not exist")
  endif()
  set(heat_source_target ${ELECTRODE_HEAT_SOURCE_TARGET})
else()
  add_library(heat-boundary-source-kernel STATIC source-kernel/src/surface_emission.cpp source-kernel/src/micro_layer.cpp)
  target_include_directories(heat-boundary-source-kernel PUBLIC source-kernel/include)
  target_compile_features(heat-boundary-source-kernel PUBLIC cxx_std_20)
  target_compile_options(heat-boundary-source-kernel PRIVATE -ffp-contract=off -fno-fast-math)
  set(heat_source_target heat-boundary-source-kernel)
endif()
add_library(electrode-heat-feedback-v2 INTERFACE)
target_include_directories(electrode-heat-feedback-v2 INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(electrode-heat-feedback-v2 INTERFACE mfem MPI::MPI_CXX ${heat_source_target})
target_compile_features(electrode-heat-feedback-v2 INTERFACE cxx_std_20)
add_executable(electrode-heat-feedback-v2-tests test_binding.cpp)
target_link_libraries(electrode-heat-feedback-v2-tests PRIVATE electrode-heat-feedback-v2)
target_compile_options(electrode-heat-feedback-v2-tests PRIVATE -ffp-contract=off -fno-fast-math)
