cmake_minimum_required(VERSION 3.20)
project(power_flow_mfem_foil_source LANGUAGES CXX)
find_package(MFEM 4.10 CONFIG REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
if(NOT MFEM_USE_MPI)
  message(FATAL_ERROR "MPI MFEM is required")
endif()
set(MFEM_SOURCE_ROOT "" CACHE PATH "External matching MFEM source")
set(COMMON "${MFEM_SOURCE_ROOT}/miniapps/common")
set(JOULE "${MFEM_SOURCE_ROOT}/miniapps/electromagnetics")
add_library(foil_joule_common STATIC
  "${COMMON}/mesh_extras.cpp" "${COMMON}/fem_extras.cpp"
  "${COMMON}/particles_extras.cpp" "${COMMON}/pfem_extras.cpp" "${COMMON}/dist_solver.cpp")
target_link_libraries(foil_joule_common PUBLIC mfem MPI::MPI_CXX)
target_compile_features(foil_joule_common PUBLIC cxx_std_17)
add_executable(power-flow-foil-joule foil_driver.cpp "${JOULE}/joule_solver.cpp")
target_include_directories(power-flow-foil-joule PRIVATE "${JOULE}")
target_link_libraries(power-flow-foil-joule PRIVATE foil_joule_common)
target_compile_features(power-flow-foil-joule PRIVATE cxx_std_17)
