cmake_minimum_required(VERSION 3.24)
project(ss304_thermoelastic_v4 LANGUAGES CXX)
find_package(MFEM 4.10 CONFIG REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
add_library(ss304-source-kernel-v4 STATIC source-kernel/src/surface_emission.cpp source-kernel/src/micro_layer.cpp)
target_include_directories(ss304-source-kernel-v4 PUBLIC source-kernel/include)
target_compile_features(ss304-source-kernel-v4 PUBLIC cxx_std_20)
target_compile_options(ss304-source-kernel-v4 PRIVATE -ffp-contract=off -fno-fast-math)
add_library(ss304-thermoelastic-v4 STATIC thermoelastic_v4.cpp)
target_include_directories(ss304-thermoelastic-v4 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(ss304-thermoelastic-v4 PUBLIC cxx_std_20)
target_compile_options(ss304-thermoelastic-v4 PRIVATE -ffp-contract=off -fno-fast-math)
target_link_libraries(ss304-thermoelastic-v4 PUBLIC mfem MPI::MPI_CXX ss304-source-kernel-v4)
add_executable(ss304-thermoelastic-v4-server thermoelastic_v4_server.cpp)
target_link_libraries(ss304-thermoelastic-v4-server PRIVATE ss304-thermoelastic-v4)
add_executable(ss304-thermoelastic-v4-tests test_thermoelastic_v4.cpp)
target_link_libraries(ss304-thermoelastic-v4-tests PRIVATE ss304-thermoelastic-v4)
target_compile_options(ss304-thermoelastic-v4-tests PRIVATE -ffp-contract=off -fno-fast-math)
# Unchanged sealed-v2 regression source compiled against the working derivative.
add_executable(ss304-sealed-heat-v2-regression sealed_v2_regression.cpp)
target_include_directories(ss304-sealed-heat-v2-regression BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ss304-sealed-heat-v2-regression PRIVATE ss304-thermoelastic-v4)
target_compile_options(ss304-sealed-heat-v2-regression PRIVATE -ffp-contract=off -fno-fast-math)
add_executable(pilot-heat-coordinates-v4 ../../tools/pilot_heat_coordinates_v4.cpp)
target_link_libraries(pilot-heat-coordinates-v4 PRIVATE mfem MPI::MPI_CXX)
target_compile_features(pilot-heat-coordinates-v4 PRIVATE cxx_std_20)
